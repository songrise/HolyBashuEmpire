hbe.100 = {
    type = country_event title = hbe.100.title desc = hbe.100.desc outcome = neutral
    illustration_tags = { 1000 = hbe_event_art_water }
    trigger = { hbe_is_prince = yes has_variable = hbe_home_chengdu_plain NOT = { has_variable = hbe_flavor_cooldown } }
    immediate = { set_variable = { name = hbe_flavor_cooldown value = yes years = 5 } }
    option = { name = hbe.100.a trigger = { var:hbe_water >= 10 } change_variable = { name = hbe_water subtract = 10 } hbe_add_credit = yes }
    option = { name = hbe.100.b trigger = { gold >= 3 } add_gold = -3 change_variable = { name = hbe_water add = 5 } hbe_add_credit = yes }
    option = { name = hbe.100.c change_variable = { name = hbe_yiqi subtract = 5 } hbe_clamp_yiqi = yes add_prestige = -1 }
}
