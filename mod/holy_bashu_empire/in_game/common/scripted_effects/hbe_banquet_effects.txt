hbe_clamp_guoqi = {
    if = { limit = { var:hbe_guoqi > 100 } set_variable = { name = hbe_guoqi value = 100 } }
    if = { limit = { var:hbe_guoqi < 0 } set_variable = { name = hbe_guoqi value = 0 } }
}
hbe_banquet_clamp = {
    if = { limit = { var:hbe_dispute > 100 } set_variable = { name = hbe_dispute value = 100 } }

    situation:hbe_banquet = {
        if = { limit = { prev = { var:hbe_delivery_kind = 1 } var:hbe_project_kind = 1 }
            if = { limit = { prev = { var:hbe_delivery_role = 1 } } change_variable = { name = hbe_role_1 add = 1 } }
            else_if = { limit = { prev = { var:hbe_delivery_role = 2 } } change_variable = { name = hbe_role_2 add = 1 } }
            else = { change_variable = { name = hbe_role_3 add = 1 } }
        }
        else_if = { limit = { prev = { var:hbe_delivery_kind = 2 } var:hbe_project_kind = 2 }
            if = { limit = { prev = { var:hbe_delivery_role = 1 } } change_variable = { name = hbe_role_1 add = 1 } }
            else_if = { limit = { prev = { var:hbe_delivery_role = 2 } } change_variable = { name = hbe_role_2 add = 1 } }
            else = { change_variable = { name = hbe_role_3 add = 1 } }
        }
    }
    if = { limit = { var:hbe_delivery_kind = 0 } international_organization:hbe_empire ?= { change_variable = { name = hbe_guoqi add = 2 } hbe_clamp_guoqi = yes } situation:hbe_banquet ?= { change_variable = { name = hbe_dispute subtract = 1 } hbe_banquet_clamp = yes } }
    trigger_event_non_silently = { id = hbe_banquet.3 }
}
hbe_settle_accounts = {
    if = { limit = { var:hbe_role_1 >= 1 var:hbe_role_2 >= 1 var:hbe_role_3 >= 1 var:hbe_dispute < 75 }
        change_variable = { name = hbe_completed_projects add = 1 } change_variable = { name = hbe_dispute subtract = 5 } international_organization:hbe_empire ?= { change_variable = { name = hbe_guoqi add = 5 } hbe_clamp_guoqi = yes }
        international_organization:hbe_empire ?= { every_international_organization_member = {
            if = { limit = { has_variable = hbe_round_delivered }
                every_owned_location = { limit = { is_in_scripted_geography = scripted_geography:hbe_domain }
                    add_location_modifier = { modifier = hbe_banquet_success years = 3 mode = replace } }
                if = { limit = { situation:hbe_banquet = { var:hbe_charter = 3 } } add_gold = 1 }
                trigger_event_non_silently = { id = hbe_banquet.4 }
            }
        } }
    }
    else = { change_variable = { name = hbe_dispute add = 8 } international_organization:hbe_empire ?= { change_variable = { name = hbe_guoqi subtract = 4 } hbe_clamp_guoqi = yes }
        international_organization:hbe_empire ?= { every_international_organization_member = { trigger_event_non_silently = { id = hbe_banquet.5 } } }
    }
    hbe_banquet_clamp = yes hbe_advance_chapter = yes
    change_variable = { name = hbe_round add = 1 }
    if = { limit = { var:hbe_project_kind = 1 } set_variable = { name = hbe_project_kind value = 2 } }
    else = { set_variable = { name = hbe_project_kind value = 1 } }
    international_organization:hbe_empire ?= { every_international_organization_member = { remove_variable = hbe_subsidized_round } }
    hbe_open_accounts = yes
}
